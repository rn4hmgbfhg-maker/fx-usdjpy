# -*- coding: utf-8 -*-
"""FXドル円 デイトレ複合時間軸 — クラウド単独実行版（Mac不要・依存ライブラリ不要）

Mac側エンジン(intraday_engine.py)と同じ判定を、公開データだけで再現する。
ポジションは価格履歴からの完全リプレイで復元する（ドンチャンは履歴だけで
現在状態が一意に決まる性質を利用）。標準ライブラリのみで動く。

パラメータはMacのconfig.jsonと同期させること（変更時はこのスクリプトも更新）:
  2Hドンチャン12/6・ATRストップ×3.0・目標利確ATR×6.0・4Hドンチャン30中心線フィルタ
  ・確認MACD(12,26,9)順行・リスク0.4%・レバ上限1.5
"""
import json
import urllib.request
from datetime import datetime, timedelta, timezone

# ==== パラメータ（Mac config.jsonの「デイトレ」と同期） ====
TF_H = 2
ENTRY_N, EXIT_N = 12, 6
ATR_K, TP_K = 3.0, 6.0
FILTER_4H_N = 30            # 4Hドンチャン中心線フィルタ期間（0=無効）
CONFIRM_MACD = True         # 確認フィルタ: MACD(12,26,9)順行ゲート
RISK, LEV = 0.004, 1.5
EQUITY, LOT = 1_000_000, 1000
TRAIL_NOTIFY_ATR = 0.5      # トレイル更新の量子化（Macエンジンと同一）
EVENT_BLOCK_MIN = 90
JST = timezone(timedelta(hours=9))
# 運用開始アンカー（この時刻より前のバーでは建玉しない＝Mac実運用と同期。
# Mac側でポジションをリセットした場合はこの日時をその時刻へ更新すること）
ANCHOR_JST = "2026-08-10 09:00"

STAR5 = ["non-farm", "average hourly earnings", "fomc",
         "federal funds rate", "cpi", "boj", "monetary policy"]


def http_json(url):
    req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
    with urllib.request.urlopen(req, timeout=30) as r:
        return json.loads(r.read().decode())


def fetch_1h():
    """Yahoo 1時間足730日（確定バーのみ）。"""
    d = http_json("https://query1.finance.yahoo.com/v8/finance/chart/"
                  "USDJPY=X?range=730d&interval=1h")["chart"]["result"][0]
    q = d["indicators"]["quote"][0]
    now = datetime.now(timezone.utc).timestamp()
    return [(t, o, h, lo, c) for t, o, h, lo, c in zip(
        d["timestamp"], q["open"], q["high"], q["low"], q["close"])
        if None not in (o, h, lo, c) and t + 3600 <= now]


def agg_bars(bars1h, hours):
    """1時間足→N時間足（UTC固定グリッド）。末尾の未完成ビンは捨てる
    （Mac data_fetch_intraday.resample と同じ確定バーのみ方針）。"""
    agg = {}
    for t, o, h, lo, c in bars1h:
        k = t // (hours * 3600) * (hours * 3600)
        if k not in agg:
            agg[k] = [o, h, lo, c]
        else:
            a = agg[k]
            a[1] = max(a[1], h)
            a[2] = min(a[2], lo)
            a[3] = c
    out = sorted((k, *v) for k, v in agg.items())
    if out and out[-1][0] + hours * 3600 > bars1h[-1][0] + 3600:
        out = out[:-1]
    return out


def trend_4h_series(bars, bars4h, n=FILTER_4H_N):
    """4Hドンチャン中心線トレンド(+1/-1/0)を執行足バーへ整列する。
    各執行バーの終了時点で「終了済みの最新4Hバー」の値を使う
    （Mac strategies_intraday.align_trend の merge_asof backward と同一）。"""
    tr4 = []
    for j in range(len(bars4h)):
        if j + 1 >= n:
            mh = max(b[2] for b in bars4h[j - n + 1:j + 1])
            ml = min(b[3] for b in bars4h[j - n + 1:j + 1])
            mid = (mh + ml) / 2
            c4 = bars4h[j][4]
            tr4.append(1 if c4 > mid else -1 if c4 < mid else 0)
        else:
            tr4.append(0)
    trend = [0] * len(bars)
    j, cur = 0, 0
    for i, b in enumerate(bars):
        te = b[0] + TF_H * 3600
        while j < len(bars4h) and bars4h[j][0] + 4 * 3600 <= te:
            cur = tr4[j]
            j += 1
        trend[i] = cur
    return trend


def macd_masks(bars, fast=12, slow=26, sig_n=9):
    """MACD順行ゲート（Mac tech_filters と同一: pandas ewm adjust=False
    相当の再帰EMA。MACD-シグナルの符号で許可方向を決める）。"""
    def ema_next(prev, x, span):
        return x if prev is None else prev + 2 / (span + 1) * (x - prev)
    ok_l = [False] * len(bars)
    ok_s = [False] * len(bars)
    e_f = e_s = sg = None
    for i, b in enumerate(bars):
        c = b[4]
        e_f = ema_next(e_f, c, fast)
        e_s = ema_next(e_s, c, slow)
        m = e_f - e_s
        sg = ema_next(sg, m, sig_n)
        d = m - sg
        ok_l[i] = d > 0
        ok_s[i] = d < 0
    return ok_l, ok_s


def atr_series(bars, n=14):
    """Wilder型ATR（ewm alpha=1/n, adjust=False）— Mac backtest.atrと同一。"""
    out, prev_c, a = [], None, None
    for _, o, h, lo, c in bars:
        tr = h - lo if prev_c is None else max(h - lo, abs(h - prev_c),
                                               abs(lo - prev_c))
        a = tr if a is None else a + (tr - a) / n
        out.append(a)
        prev_c = c
    return out


def replay(bars, atr, trend, cf_l, cf_s):
    """ライブエンジンと同じ規則でポジションを復元する。
    シグナルバー引けで建玉（entry=close）・ATR×3ストップ・ATR×6利確、
    以降の確定バーでストップ/利確ヒット（ストップ優先）とトレイル
    （0.5ATR以上の改善のみ反映＝Macの通知量子化と同一）。
    trend=4Hフィルタ系列・cf_l/cf_s=MACD確認ゲート（新規建て・ドテンのみ
    選別し、手仕舞いには関与しない＝Mac donchian_mtf と同一）。"""
    hi = [None] * len(bars)
    lo = [None] * len(bars)
    xhi = [None] * len(bars)
    xlo = [None] * len(bars)
    for i in range(len(bars)):
        if i >= ENTRY_N:
            hi[i] = max(b[2] for b in bars[i - ENTRY_N:i])
            lo[i] = min(b[3] for b in bars[i - ENTRY_N:i])
        if i >= EXIT_N:
            xhi[i] = max(b[2] for b in bars[i - EXIT_N:i])
            xlo[i] = min(b[3] for b in bars[i - EXIT_N:i])

    # --- 第1層: 全履歴のドンチャン状態系列（Mac donchian_mtfと同一）
    #     Macエンジンはこの系列を参照するため、稼働開始時に途中トレンドへ乗る
    sig_series = [0] * len(bars)
    spos = 0
    for i in range(len(bars)):
        c = bars[i][4]
        if hi[i] is None:
            continue

        def ok(d):
            if FILTER_4H_N and trend[i] != d:
                return False
            if CONFIRM_MACD:
                return cf_l[i] if d > 0 else cf_s[i]
            return True

        if spos == 0:
            if c > hi[i] and ok(1):
                spos = 1
            elif c < lo[i] and ok(-1):
                spos = -1
        elif spos == 1 and c < xlo[i]:
            spos = -1 if (c < lo[i] and ok(-1)) else 0
        elif spos == -1 and c > xhi[i]:
            spos = 1 if (c > hi[i] and ok(1)) else 0
        sig_series[i] = spos

    # --- 第2層: 運用リプレイ（アンカー以降のみ建玉。Mac intraday_engineと同一）
    anchor_ts = datetime.strptime(ANCHOR_JST, "%Y-%m-%d %H:%M") \
        .replace(tzinfo=JST).timestamp() - TF_H * 3600   # バー開始時刻に変換
    pos = 0
    st = {}
    last_event = "待機"
    for i in range(len(bars)):
        t, o, h, lo_, c = bars[i]
        if hi[i] is None or t < anchor_ts:
            continue
        # 1) 保有中: ストップ/利確ヒット（ストップ優先・確定バー高安）
        if pos != 0:
            if (pos > 0 and lo_ <= st["stop"]) or \
               (pos < 0 and h >= st["stop"]):
                pos = 0
                st = {}
                last_event = "ストップ決済"
            elif (pos > 0 and h >= st["tp"]) or (pos < 0 and lo_ <= st["tp"]):
                pos = 0
                st = {}
                last_event = "利確決済"
        # 2) シグナルと運用ポジの突合（引け建て・entry=close）
        sig = sig_series[i]
        if sig != pos:
            pos = sig
            if pos != 0:
                a = atr[i]
                st = {"entry": c, "best": c,
                      "stop": round(c - pos * ATR_K * a, 3),
                      "tp": round(c + pos * TP_K * a, 3),
                      "entry_time": t, "atr0": a}
                last_event = "新規" + ("買" if pos > 0 else "売")
            else:
                st = {}
                last_event = "シグナル決済"
        elif pos != 0:
            # 3) トレイル（0.5ATR以上の改善のみ反映＝Mac通知量子化と同一）
            st["best"] = max(st["best"], c) if pos > 0 else min(st["best"], c)
            cand = st["best"] - pos * ATR_K * atr[i]
            better = (cand - st["stop"]) * pos
            if better >= TRAIL_NOTIFY_ATR * atr[i]:
                st["stop"] = round(cand, 3)
    return pos, st, last_event


def _feed(url):
    """Forex Factoryフィードを3時間キャッシュ付きで取る。

    ★2026-08-16 追加: キャッシュ無しで毎回叩くと配信元が HTTP 429 を返し、
    「カレンダー取得失敗」が常態化する（実測で確認）。Mac側 events.py と同じ
    3時間フレッシュ／72時間ステールの二段構えにする。保存形式もMacと同一
    （{"fetched": iso, "data": [...]}）なので results/cache/*.json をそのまま
    コピーして流用できる。毎時実行する本システムでは必須の対策。"""
    import hashlib
    import os
    d = os.path.join(os.path.expanduser("~"), ".fx_ff_cache")
    p = os.path.join(d, hashlib.md5(url.encode()).hexdigest() + ".json")

    def read(max_age_h):
        try:
            with open(p, encoding="utf-8") as f:
                obj = json.load(f)
            if isinstance(obj, list):
                obj = {"fetched": None, "data": obj}
            age = (datetime.now(timezone.utc)
                   - datetime.fromisoformat(obj["fetched"])).total_seconds() \
                if obj.get("fetched") else \
                datetime.now().timestamp() - os.path.getmtime(p)
            return obj["data"] if age <= max_age_h * 3600 else None
        except Exception:
            return None

    fresh = read(3)
    if fresh is not None:
        return fresh
    try:
        data = http_json(url)
        os.makedirs(d, exist_ok=True)
        with open(p, "w", encoding="utf-8") as f:
            json.dump({"fetched": datetime.now(timezone.utc).isoformat(),
                       "data": data}, f)
        return data
    except Exception:
        return read(72)


def events_24h():
    ev, ok = [], False
    now = datetime.now(timezone.utc)
    for u in ("https://nfs.faireconomy.media/ff_calendar_thisweek.json",
              "https://nfs.faireconomy.media/ff_calendar_nextweek.json"):
        data = _feed(u)
        if data is None:
            continue
        ok = True
        for e in data:
            if e.get("impact") != "High" or e.get("country") not in ("USD",
                                                                    "JPY"):
                continue
            if not any(p in e.get("title", "").lower() for p in STAR5):
                continue
            try:
                dt = datetime.fromisoformat(e["date"])
            except (KeyError, ValueError):
                continue
            if now <= dt <= now + timedelta(hours=24):
                ev.append((dt, e["country"], e.get("title", "")))
    return (sorted(set(ev)) if ok else None)


def main():
    bars1h = fetch_1h()
    bars = agg_bars(bars1h, TF_H)
    bars4h = agg_bars(bars1h, 4)
    atr = atr_series(bars)
    trend = trend_4h_series(bars, bars4h)
    cf_l, cf_s = macd_masks(bars)
    pos, st, last_event = replay(bars, atr, trend, cf_l, cf_s)
    t, o, h, lo_, c = bars[-1]
    bar_jst = datetime.fromtimestamp(t + TF_H * 3600, JST)
    a = atr[-1]
    now = datetime.now(JST)

    # ★2026-08-16 追加: データ鮮度ガード（Mac intraday15_engine と同じ考え方）。
    # Yahoo障害・レート制限中に過去のブレイクを「今のシグナル」として通知する
    # 事故を防ぐ。閾値 = 時間足の2本分+30分。週末・祝日の休場中は必ず該当する
    # が、それは正常（新規判定を出さないだけ）。
    age_min = (now.timestamp() - (t + TF_H * 3600)) / 60
    stale = age_min > TF_H * 120 + 30

    ev = events_24h()
    soon = [] if ev is None else \
        [e for e in ev if (e[0] - datetime.now(timezone.utc)).total_seconds()
         <= EVENT_BLOCK_MIN * 60]

    L = [f"■ ドル円 デイトレ指示書【クラウド代行】 {now:%Y-%m-%d %H:%M}",
         f"  確定バー: {bar_jst:%Y-%m-%d %H:%M}（{TF_H}時間足）"
         f"  終値: {c:.3f}円  ATR(14): {a:.3f}円",
         "-" * 46,
         f"▼ デイトレ複合時間軸（{TF_H}Hドンチャン{ENTRY_N}/{EXIT_N}"
         f"・ATR×{ATR_K}・利確ATR×{TP_K}"
         f"・4Hフィルタ{FILTER_4H_N if FILTER_4H_N else '無'}"
         + ("・確認MACD順行" if CONFIRM_MACD else "")
         + f"・リスク{RISK:.1%}）"]

    if stale:
        L.append(f"⚠データ鮮度不足: 最終確定バーが{age_min:.0f}分前。"
                 f"新規判定は見送る（休場中なら正常。平日ならYahoo障害/"
                 f"レート制限を疑う）。以下の現況は参考値。")

    if pos != 0:
        et = datetime.fromtimestamp(st["entry_time"] + TF_H * 3600, JST)
        L += [f"【現況】{'買' if pos > 0 else '売'}ポジション保有"
              f"（{et:%m/%d %H:%M}バー建て・建値{st['entry']:.3f}円）",
              f"  決済逆指値: {st['stop']:.3f}円（トレイリング）",
              f"  目標利確指値: {st['tp']:.3f}円（固定）",
              f"  数量目安: 建玉時ATR{st['atr0']:.3f}円 → "
              f"{int(min(EQUITY * RISK / (ATR_K * st['atr0']), EQUITY * LEV / st['entry']) // LOT * LOT):,}通貨",
              "  → 口座の建玉・OCOが上記と一致しているか確認。"
              "乖離があれば上記へ合わせる。"]
    else:
        L += [f"【待機】ノーポジ（直近イベント: {last_event}）。",

## 保守（重要）

- **パラメータ同期**: 日次研究がパラメータを進化させたら、スクリプト冒頭の定数
  （`TF_H` / `ENTRY_N` / `EXIT_N` / `ATR_K` / `TP_K` / `FILTER_4H_N` / `CONFIRM_MACD`）を
  `config.json` の「デイトレ」に合わせて更新する。
- **アンカー同期**: Mac側でポジションをリセット（手動フラット化・スキーム再起動）したら、
  `ANCHOR_JST` をその日時（次に建玉可能になるバーの終了時刻）へ更新する。
- **発注方式**: 新規＝成行＋決済OCO（ATR×3.0トレイリング逆指値＋ATR×6.0目標利確指値）。
  トレイルで逆指値のみ動き、利確指値は建玉時のまま固定。
- **トレイルの量子化**: 本システムは `TRAIL_NOTIFY_ATR = 0.5`（第4システムは1.0）。
  ATR×0.5以上の改善があった時だけ逆指値を動かす＝Macエンジンの通知量子化と同一。
